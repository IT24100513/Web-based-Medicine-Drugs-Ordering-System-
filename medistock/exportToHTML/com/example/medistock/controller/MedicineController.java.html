1    package com.example.medistock.controller;
2    
3    import com.example.medistock.model.Medicine;
4    import com.example.medistock.repository.MedicineRepository;
5    import org.springframework.beans.factory.annotation.Autowired;
6    import org.springframework.web.bind.annotation.*;
7    
8    import java.util.List;
9    
10   @CrossOrigin(origins = "*") // To allow frontend to access
11   @RestController
12   @RequestMapping("/api/medicines")
13   public class MedicineController {
14   
15       @Autowired
16       private MedicineRepository repo;
17   
18       @GetMapping
19       public List<Medicine> getAll() {
20           return repo.findAll();
21       }
22   
23       @PostMapping
24       public Medicine addMedicine(@RequestBody Medicine med) {
25           return repo.save(med);
26       }
27   
28       @DeleteMapping("/{id}")
29       public void deleteMedicine(@PathVariable Long id) {
30           repo.deleteById(id);
31       }
32   
33       @PutMapping("/{id}")
34       public Medicine updateMedicine(@PathVariable Long id, @RequestBody Medicine updated) {
35           Medicine med = repo.findById(id).orElseThrow();
36           med.setName(updated.getName());
37           med.setQuantity(updated.getQuantity());
38           med.setUnitPrice(updated.getUnitPrice());
39           med.setCategory(updated.getCategory());
40           med.setExpiry(updated.getExpiry());
41           med.setBatch(updated.getBatch());
42           return repo.save(med);
43       }
44   }
