238      // DOM elements
239      const inventoryList = document.getElementById('inventoryList');
240      const addItemModal = document.getElementById('addItemModal');
241      const addItemBtn = document.getElementById('addItemBtn');
242      const cancelBtn = document.getElementById('cancelBtn');
243      const addItemForm = document.getElementById('addItemForm');
244      const searchInput = document.getElementById('searchInput');
245      const categoryFilter = document.getElementById('categoryFilter');

247      // Utility functions
248      function getStockStatus(quantity) {
249          if (quantity <= 20) return { class: 'status-critical', text: 'Critical', color: 'text-red-600' };
250          if (quantity <= 50) return { class: 'status-low', text: 'Low', color: 'text-orange-600' };
251          if (quantity <= 100) return { class: 'status-medium', text: 'Medium', color: 'text-blue-600' };
252          return { class: 'status-high', text: 'Good', color: 'text-green-600' };
253      }
254  
255      function getDaysUntilExpiry(expiryDate) {
256          const today = new Date();
257          const expiry = new Date(expiryDate);
258          const diffTime = expiry - today;
259          return Math.ceil(diffTime / (1000 * 60 * 60 * 24));
260      }
261  
262      function formatCurrency(amount) {
263          return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(amount);
264      }
265  
266      // Render inventory
267      function renderInventory(items = inventory) {
268          inventoryList.innerHTML = items.map(item => {
269              const status = getStockStatus(item.quantity);
270              const daysUntilExpiry = getDaysUntilExpiry(item.expiry);
271              const totalValue = item.quantity * item.unitPrice;
272  
273              return `
274                  <div class="border border-gray-200 rounded-lg p-4 hover:shadow-md transition-shadow">
275                      <div class="flex items-center justify-between">
276                          <div class="flex-1">
277                              <div class="flex items-center space-x-3">
278                                  <h4 class="font-semibold text-gray-900">${item.name}</h4>
279                                  <span class="px-2 py-1 text-xs font-medium ${status.class} ${status.color} rounded-full">
280                                      ${status.text}
281                                  </span>
282                                  ${daysUntilExpiry <= 30 ? `<span class="px-2 py-1 text-xs font-medium bg-red-100 text-red-600 rounded-full">Expires in ${daysUntilExpiry} days</span>` : ''}
283                              </div>
284                              <div class="mt-2 grid grid-cols-2 md:grid-cols-4 gap-4 text-sm text-gray-600">
285                                  <div>
286                                      <span class="font-medium">Quantity:</span> ${item.quantity} units
287                                  </div>
288                                  <div>
289                                      <span class="font-medium">Unit Price:</span> ${formatCurrency(item.unitPrice)}
290                                  </div>
291                                  <div>
292                                      <span class="font-medium">Total Value:</span> ${formatCurrency(totalValue)}
293                                  </div>
294                                  <div>
295                                      <span class="font-medium">Batch:</span> ${item.batch}
296                                  </div>
297                              </div>
298                              <div class="mt-1 text-xs text-gray-500">
299                                  Category: ${item.category.charAt(0).toUpperCase() + item.category.slice(1)} | Expires: ${item.expiry}
300                              </div>
301                          </div>
302                          <div class="flex space-x-2 ml-4">
303                              <button onclick="editItem(${item.id})" class="px-3 py-1 bg-blue-100 text-blue-600 rounded-lg hover:bg-blue-200 transition-colors text-sm">
304                                  Edit
305                              </button>
306                              <button onclick="deleteItem(${item.id})" class="px-3 py-1 bg-red-100 text-red-600 rounded-lg hover:bg-red-200 transition-colors text-sm">
307                                  Delete
308                              </button>
309                          </div>
310                      </div>
311                  </div>
312              `;
313          }).join('');
314      }
315  
316      // Filter and search functionality
317      function filterInventory() {
318          const searchTerm = searchInput.value.toLowerCase();
319          const selectedCategory = categoryFilter.value;
320  
321          const filtered = inventory.filter(item => {
322              const matchesSearch = item.name.toLowerCase().includes(searchTerm);
323              const matchesCategory = !selectedCategory || item.category === selectedCategory;
324              return matchesSearch && matchesCategory;
325          });
326  
327          renderInventory(filtered);
328      }

337          addItemModal.classList.add('hidden');
338          addItemModal.classList.remove('flex');

373      // Item actions
374      function editItem(id) {
375          const item = inventory.find(i => i.id === id);
376          if (item) {
377              const newQuantity = prompt(`Edit quantity for ${item.name}:`, item.quantity);
378              if (newQuantity && !isNaN(newQuantity)) {
379                  item.quantity = parseInt(newQuantity);
380                  renderInventory();
381              }
382          }
383      }
384  
385      function deleteItem(id) {
386          if (confirm('Are you sure you want to delete this item?')) {
387              inventory = inventory.filter(i => i.id !== id);
388              renderInventory();
389          }
390      }
